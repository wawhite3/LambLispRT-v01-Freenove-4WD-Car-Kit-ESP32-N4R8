1205inline int ll_u8_seqlen(unsigned char b0)
1206{
1207 if (b0 < 0x80) return 1;
1208 if ((b0 & 0xE0) == 0xC0) return 2;
1209 if ((b0 & 0xF0) == 0xE0) return 3;
1210 if ((b0 & 0xF8) == 0xF0) return 4;
1211 return 1;
1212}
1213
1216inline ll_codepoint_t ll_u8_decode(const char *p, int *nb)
1217{
1218 unsigned char b0 = (unsigned char) p[0];
1219 int n = ll_u8_seqlen(b0);
1220 if (nb) *nb = n;
1221 if (n == 1) return (ll_codepoint_t) b0;
1222 ll_codepoint_t cp = b0 & (0xFF >> (n + 1));
1223 for (int i = 1; i < n; i++) {
1224 unsigned char c = (unsigned char) p[i];
1225 if ((c & 0xC0) != 0x80) { if (nb) *nb = 1; return (ll_codepoint_t) b0; }
1226 cp = (cp << 6) | (c & 0x3F);
1227 }
1228 return cp;
1229}
1230
1232inline int ll_u8_encode(ll_codepoint_t cp, char *out)
1233{
1234 if (cp < 0x80) { out[0] = (char) cp; return 1; }
1235 if (cp < 0x800) { out[0] = (char)(0xC0 | (cp >> 6)); out[1] = (char)(0x80 | (cp & 0x3F)); return 2; }
1236 if (cp < 0x10000) { out[0] = (char)(0xE0 | (cp >> 12)); out[1] = (char)(0x80 | ((cp >> 6) & 0x3F));
1237 out[2] = (char)(0x80 | (cp & 0x3F)); return 3; }
1238 out[0] = (char)(0xF0 | (cp >> 18)); out[1] = (char)(0x80 | ((cp >> 12) & 0x3F));
1239 out[2] = (char)(0x80 | ((cp >> 6) & 0x3F)); out[3] = (char)(0x80 | (cp & 0x3F)); return 4;
1240}

1343 char *dec(LL_int64 n) {
1344 // Full 64-bit signed decimal; avoids conflict with dec(Word_t) on 64-bit platforms.
1345 LL_int32 ix = buffer_size - 1;
1346 char *ptr = &(buffer[ix]);
1347 *ptr = '\0';
1348 bool neg = (n < 0);
1349 uint64_t uv = neg ? (n == INT64_MIN ? (uint64_t) 9223372036854775808ULL : (uint64_t) -n)
1350 : (uint64_t) n;
1351 do { *--ptr = (char)('0' + (uv % 10)); uv /= 10; } while (uv);
1352 if (neg) *--ptr = '-';
1353 return ptr;
1354 }

1356 char *hex(Word_t n) {
1357 LL_int32 ix = buffer_size - 1;
1358 char *ptr = &(buffer[ix]);
1359 LL_int32 nibs = sizeof(Word_t) * 2;
1360
1361 *ptr = '\0';
1362 while (nibs--) {
1363 Word_t d = n & 0x0f;
1364 n >>= 4;
1365 char c = (d < 10) ? (d + '0') : (d - 10 + 'a');
1366 *--ptr = c;
1367 }
1368 return ptr;
