140{
141 ll_float_prof[LL_FLOAT_PROF_TOTAL]++;

143 if (v == 1.0 / 0.0 || v == -1.0 / 0.0) { ll_float_prof[LL_FLOAT_PROF_SPEC]++; return; }

817 LL_float64 coerce_float64()
818 {
819 ME("Cell::coerce_float64()");
820 LL_int32 typ = type();
821 if (typ == T_FLOAT64) return as_float64();
822 else if (typ == T_FLOAT32) return (LL_float64) as_float32();
823 else if (typ == T_INT32) return (LL_float64) as_int32();
824 else if (typ == T_INT64) return (LL_float64) as_int64();
825 else if (typ == T_RATIONAL) {

830 else if (typ == T_BIGNUM) {
831 // limb-by-limb: payload = [int32 sign_nlimbs][LL_int32 limb0 ... limbN-1], little-endian.
832 const int32_t *pay = (const int32_t *) as_ByteVec_t();
